(defun my/add-to-path-if-exists (dir)
  "Add DIR to `PATH' and `exec-path' if DIR exists."
  (let ((dir (expand-file-name dir)))
    (when (file-directory-p dir)
      (setenv "PATH" (concat dir path-separator (getenv "PATH")))
      (add-to-list 'exec-path dir))))

* ELPACA

GitHub repo: https://github.com/progfolio/elpaca, we just follow the instructions here:

#+begin_src emacs-lisp
(defvar elpaca-installer-version 0.12)
(defvar elpaca-directory (expand-file-name "elpaca/" user-emacs-directory))
(defvar elpaca-builds-directory (expand-file-name "builds/" elpaca-directory))
(defvar elpaca-sources-directory (expand-file-name "sources/" elpaca-directory))
(defvar elpaca-order '(elpaca :repo "https://github.com/progfolio/elpaca.git"
                              :ref nil :depth 1 :inherit ignore
                              :files (:defaults "elpaca-test.el" (:exclude "extensions"))
                              :build (:not elpaca-activate)))
(let* ((repo  (expand-file-name "elpaca/" elpaca-sources-directory))
       (build (expand-file-name "elpaca/" elpaca-builds-directory))
       (order (cdr elpaca-order))
       (default-directory repo))
  (add-to-list 'load-path (if (file-exists-p build) build repo))
  (unless (file-exists-p repo)
    (make-directory repo t)
    (when (<= emacs-major-version 28) (require 'subr-x))
    (condition-case-unless-debug err
        (if-let* ((buffer (pop-to-buffer-same-window "*elpaca-bootstrap*"))
                  ((zerop (apply #'call-process `("git" nil ,buffer t "clone"
                                                  ,@(when-let* ((depth (plist-get order :depth)))
                                                      (list (format "--depth=%d" depth) "--no-single-branch"))
                                                  ,(plist-get order :repo) ,repo))))
                  ((zerop (call-process "git" nil buffer t "checkout"
                                        (or (plist-get order :ref) "--"))))
                  (emacs (concat invocation-directory invocation-name))
                  ((zerop (call-process emacs nil buffer nil "-Q" "-L" "." "--batch"
                                        "--eval" "(byte-recompile-directory \".\" 0 'force)")))
                  ((require 'elpaca))
                  ((elpaca-generate-autoloads "elpaca" repo)))
            (progn (message "%s" (buffer-string)) (kill-buffer buffer))
          (error "%s" (with-current-buffer buffer (buffer-string))))
      ((error) (warn "%s" err) (delete-directory repo 'recursive))))
  (unless (require 'elpaca-autoloads nil t)
    (require 'elpaca)
    (elpaca-generate-autoloads "elpaca" repo)
    (let ((load-source-file-function nil)) (load "./elpaca-autoloads"))))
(add-hook 'after-init-hook #'elpaca-process-queues)
(elpaca `(,@elpaca-order))

* BACKUP

By default, Emacs creates automatic backups of files in their original
such, directories "file.el" and the backup "file.el~".  This leads to

* VTERM

#+begin_src emacs-lisp
(defvar-local my/vterm-kitty-dark-p nil
  "Non-nil means this vterm buffer should use the Kitty dark palette.")

(defun my/vterm-kitty-dark-get-color (orig-fun index &rest args)
  "Return Kitty dark default colors for vterm buffers.

This fixes the actual vterm background, unlike `face-remap-add-relative'
or `buffer-face-mode', which vterm's default-color path does not honor."
  (if (and (bound-and-true-p my/vterm-kitty-dark-p)
           (= index -1))
      (cond
       ;; Default foreground.
       ((member :foreground args)
        "#f0f3f6")

       ;; Default background.
       (t
        "#05070a"))
    (apply orig-fun index args)))

(defun my/vterm-kitty-dark-mode ()
  "Use Kitty-like dark colors in the current vterm buffer."
  (setq-local my/vterm-kitty-dark-p t)

  ;; Normal Emacs buffer background fallback. This alone is not enough for
  ;; vterm, but it helps with margins/edge cases.
  (setq-local buffer-face-mode-face
              '(:foreground "#f0f3f6"
                :background "#05070a"))
  (buffer-face-mode 1)

  ;; Selection.
  (face-remap-add-relative
   'region
   '(:foreground "#ffffff"
     :background "#1f6feb"))

  ;; Cursor.
  (setq-local cursor-type 'box)

  ;; No line numbers in terminals.
  (display-line-numbers-mode -1))

(use-package vterm
  :ensure t
  :defer t
  :commands vterm
  :hook
  (vterm-mode . my/vterm-kitty-dark-mode)
  :config
  (setq vterm-max-scrollback 5000
        vterm-shell "/bin/bash"
        shell-file-name "/bin/bash"
        vterm-timer-delay 0.01)

  ;; This is the important part. It makes vterm's own default background dark.
  (advice-add 'vterm--get-color :around #'my/vterm-kitty-dark-get-color)

  ;; Kitty palette: color0..color15.
  ;;
  ;; vterm documents these as the faces to customize for its terminal colors.
  ;; They affect ANSI colors, not the default background.
  (set-face-attribute 'vterm-color-black nil
                      :foreground "#303030"
                      :background "#303030")
  (set-face-attribute 'vterm-color-red nil
                      :foreground "#ff6b7a"
                      :background "#ff6b7a")
  (set-face-attribute 'vterm-color-green nil
                      :foreground "#4ade80"
                      :background "#4ade80")
  (set-face-attribute 'vterm-color-yellow nil
                      :foreground "#facc15"
                      :background "#facc15")
  (set-face-attribute 'vterm-color-blue nil
                      :foreground "#60a5fa"
                      :background "#60a5fa")
  (set-face-attribute 'vterm-color-magenta nil
                      :foreground "#c084fc"
                      :background "#c084fc")
  (set-face-attribute 'vterm-color-cyan nil
                      :foreground "#22d3ee"
                      :background "#22d3ee")
  (set-face-attribute 'vterm-color-white nil
                      :foreground "#c9d1d9"
                      :background "#c9d1d9")

  (set-face-attribute 'vterm-color-bright-black nil
                      :foreground "#8b949e"
                      :background "#8b949e")
  (set-face-attribute 'vterm-color-bright-red nil
                      :foreground "#ff4d6d"
                      :background "#ff4d6d")
  (set-face-attribute 'vterm-color-bright-green nil

(defvar my/default-cursor-color nil
  "Cursor color before entering vterm.")

(defvar my/vterm-cursor-color "#ffffff"
  "Cursor color to use in vterm buffers.")

(defun my/vterm-visible-cursor ()
  "Make the cursor more visible in vterm."
  ;; This is buffer-local and safe.
  (setq-local cursor-type 'box))

(defun my/update-cursor-color-for-vterm ()
  "Use a bright cursor in vterm, restore normal cursor elsewhere."
  (unless my/default-cursor-color
    (setq my/default-cursor-color
          (frame-parameter nil 'cursor-color)))

  (if (derived-mode-p 'vterm-mode)
      (set-cursor-color my/vterm-cursor-color)
    (when my/default-cursor-color
      (set-cursor-color my/default-cursor-color))))

(add-hook 'vterm-mode-hook #'my/vterm-visible-cursor)
(add-hook 'buffer-list-update-hook #'my/update-cursor-color-for-vterm)
(add-hook 'window-selection-change-functions
          (lambda (_frame)
            (my/update-cursor-color-for-vterm)))

#+begin_src emacs-lisp
(when (eq system-type 'darwin)
  ;; Use Command as Emacs Meta
  (setq ns-command-modifier 'meta
